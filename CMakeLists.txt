cmake_minimum_required(VERSION 3.21)
project(LitzSim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
add_library(litzcore src/circuit.cpp src/solver.cpp src/waves.cpp src/pfc.cpp)
target_include_directories(litzcore PUBLIC src)
target_link_libraries(litzcore PUBLIC Qt6::Core)
add_executable(litzsim-cli src/cli.cpp src/web-command.cpp)
enable_testing()
add_executable(circuit-test tests/circuit.cpp)
target_link_libraries(circuit-test PRIVATE litzcore)
add_test(NAME circuit COMMAND circuit-test ${CMAKE_CURRENT_SOURCE_DIR}/examples)
add_executable(linear-test tests/linear.cpp)
target_link_libraries(linear-test PRIVATE litzcore)
add_test(NAME linear COMMAND linear-test ${CMAKE_CURRENT_SOURCE_DIR}/examples)
add_executable(buck-test tests/buck.cpp)
target_link_libraries(buck-test PRIVATE litzcore)
add_test(NAME buck COMMAND buck-test ${CMAKE_CURRENT_SOURCE_DIR}/examples/buck.litzsim)
add_executable(wave-test tests/waves.cpp)
target_link_libraries(wave-test PRIVATE litzcore)
add_test(NAME waves COMMAND wave-test)
add_library(litzstudio src/studio.cpp src/schematic.cpp src/viewer.cpp)
qt_add_resources(litzstudio typography PREFIX "/fonts" BASE assets/fonts
    FILES assets/fonts/PretendardVariable.ttf assets/fonts/Pretendard-OFL.txt)
qt_add_resources(litzstudio symbols PREFIX "/icons" BASE assets/icons FILES
    assets/icons/litzio.ico assets/icons/Files.svg assets/icons/LayoutGrid.svg assets/icons/X.svg
    assets/icons/Folder.svg assets/icons/FolderOpen.svg assets/icons/File.svg assets/icons/Play.svg
    assets/icons/Square.svg assets/icons/Settings2.svg assets/icons/Lucide-LICENSE.txt assets/icons/Litzio-LICENSE.txt)
target_link_libraries(litzstudio PUBLIC litzcore Qt6::Widgets PRIVATE Qt6::Svg)
target_link_libraries(litzsim-cli PRIVATE litzstudio)
add_executable(litzsim src/main.cpp)
target_link_libraries(litzsim PRIVATE litzstudio)
add_executable(studio-test tests/studio.cpp)
target_link_libraries(studio-test PRIVATE litzstudio Qt6::Svg)
add_dependencies(studio-test litzsim-cli)
add_test(NAME studio COMMAND studio-test ${CMAKE_CURRENT_SOURCE_DIR}/examples)
set_tests_properties(studio PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(viewer-test tests/viewer.cpp)
target_link_libraries(viewer-test PRIVATE litzstudio)
add_test(NAME viewer COMMAND viewer-test ${CMAKE_CURRENT_SOURCE_DIR}/examples)
set_tests_properties(viewer PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(pfc-test tests/pfc.cpp)
target_link_libraries(pfc-test PRIVATE litzcore)
add_test(NAME pfc COMMAND pfc-test ${CMAKE_CURRENT_SOURCE_DIR}/examples/pfc.litzsim)
add_executable(pfc-studio-test tests/pfc-studio.cpp)
target_link_libraries(pfc-studio-test PRIVATE litzstudio)
add_test(NAME pfc-studio COMMAND pfc-studio-test ${CMAKE_CURRENT_SOURCE_DIR}/examples/pfc.litzsim)
set_tests_properties(pfc-studio PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(workflow-test tests/workflow.cpp)
target_link_libraries(workflow-test PRIVATE litzstudio Qt6::Test)
add_test(NAME workflow COMMAND workflow-test ${CMAKE_CURRENT_SOURCE_DIR}/examples)
set_tests_properties(workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 40)

if(WIN32 AND MINGW)
    include(cmake/WindowsRuntime.cmake)
endif()
