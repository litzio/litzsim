cmake_minimum_required(VERSION 3.21)
project(LitzSim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_library(litzcore src/circuit.cpp src/solver.cpp)
target_include_directories(litzcore PUBLIC src)
target_link_libraries(litzcore PUBLIC Qt6::Core)
add_executable(litzsim-cli src/cli.cpp)
target_link_libraries(litzsim-cli PRIVATE litzcore)
enable_testing()
add_executable(circuit-test tests/circuit.cpp)
target_link_libraries(circuit-test PRIVATE litzcore)
add_test(NAME circuit COMMAND circuit-test ${CMAKE_CURRENT_SOURCE_DIR}/examples)
add_executable(linear-test tests/linear.cpp)
target_link_libraries(linear-test PRIVATE litzcore)
add_test(NAME linear COMMAND linear-test ${CMAKE_CURRENT_SOURCE_DIR}/examples)
